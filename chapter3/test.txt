## **CURRENT INFRASTRUCTURE (Already Deployed)**

### **Existing Docker Stack** (`~/citadelnms-kafka-lab/`)

**10 Running Containers:**
- `citadelnms-core` - Main NMS engine (image: `citadelnms:latest`)
- `citadel-minion-1` through `citadel-minion-5` - Distributed collectors (image: `citadelnms/minion:latest`)

### **3. Updated generate-compose.sh**
#!/usr/bin/env bash
# =============================================================================
# generate-compose.sh — Regenerate docker-compose.yml for CitadelNMS
#
# Usage:
#   ./scripts/generate-compose.sh [BROKER_COUNT] [MINION_COUNT]
#
# Examples:
#   ./scripts/generate-compose.sh          # Uses .env values (default)
#   ./scripts/generate-compose.sh 5 4      # 5 Kafka brokers, 4 Minions
# =============================================================================
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
ROOT_DIR="$(dirname "$SCRIPT_DIR")"

# --- Load .env defaults ------------------------------------------------------
set -a
source "${ROOT_DIR}/.env"
set +a

# --- Override from CLI args --------------------------------------------------
BROKER_COUNT="${1:-${BROKER_COUNT:-3}}"
MINION_COUNT="${2:-${MINION_COUNT:-2}}"

echo "============================================================"
echo " Generating docker-compose for CitadelNMS:"
echo "   Kafka brokers : $BROKER_COUNT"
echo "   Minions       : $MINION_COUNT"
echo "   Kafka version : ${KAFKA_VERSION:-3.7.0}"
echo "   Minion version: ${MINION_VERSION:-latest}"
echo "   CitadelNMS ver: ${CITADELNMS_VERSION:-latest}"
echo "============================================================"

# --- Update .env -------------------------------------------------------------
sed -i "s/^BROKER_COUNT=.*/BROKER_COUNT=${BROKER_COUNT}/" "${ROOT_DIR}/.env"
sed -i "s/^MINION_COUNT=.*/MINION_COUNT=${MINION_COUNT}/" "${ROOT_DIR}/.env"

# --- Build quorum voters string (all brokers are controllers) ----------------
VOTERS=""
for i in $(seq 1 "${BROKER_COUNT}"); do
  if [ -n "$VOTERS" ]; then VOTERS="${VOTERS},"; fi
  VOTERS="${VOTERS}${i}@citadel-kafka-${i}:9093"
done

# --- Generate Kafka broker configs -------------------------------------------
echo "[1/3] Generating Kafka broker configs..."
for i in $(seq 1 "${BROKER_COUNT}"); do
  CFG="${ROOT_DIR}/configs/kafka/broker-${i}.properties"
  INTERNAL_PORT=9092
  EXTERNAL_PORT=9094

  cat > "${CFG}" << EOF
# Auto-generated by generate-compose.sh — CitadelNMS broker ${i} of ${BROKER_COUNT}
process.roles=broker,controller
node.id=${i}
controller.quorum.voters=${VOTERS}

listeners=INTERNAL://0.0.0.0:${INTERNAL_PORT},EXTERNAL://0.0.0.0:${EXTERNAL_PORT},CONTROLLER://0.0.0.0:9093
advertised.listeners=INTERNAL://citadel-kafka-${i}:${INTERNAL_PORT},EXTERNAL://citadel-kafka-${i}:${EXTERNAL_PORT}
listener.security.protocol.map=INTERNAL:PLAINTEXT,EXTERNAL:PLAINTEXT,CONTROLLER:PLAINTEXT
inter.broker.listener.name=INTERNAL
controller.listener.names=CONTROLLER

log.dirs=/var/lib/kafka/data

default.replication.factor=$(( BROKER_COUNT < 3 ? BROKER_COUNT : 2 ))
offsets.topic.replication.factor=$(( BROKER_COUNT < 3 ? BROKER_COUNT : 2 ))
transaction.state.log.replication.factor=$(( BROKER_COUNT < 3 ? BROKER_COUNT : 2 ))
transaction.state.log.min.isr=1
min.insync.replicas=1
num.partitions=${KAFKA_PARTITIONS:-12}

num.network.threads=8
num.io.threads=12
socket.send.buffer.bytes=1048576
socket.receive.buffer.bytes=1048576
message.max.bytes=5000000
replica.fetch.max.bytes=5000000

auto.create.topics.enable=true
delete.topic.enable=false
compression.type=producer
log.retention.hours=168
EOF
  chmod 644 "${CFG}"
done

# --- Generate Minion config directories --------------------------------------
echo "[2/3] Generating Minion config directories..."

# Build bootstrap.servers strings
INTERNAL_BOOTSTRAP=""
EXTERNAL_BOOTSTRAP=""
for i in $(seq 1 "${BROKER_COUNT}"); do
  if [ -n "$INTERNAL_BOOTSTRAP" ]; then
    INTERNAL_BOOTSTRAP="${INTERNAL_BOOTSTRAP},"
    EXTERNAL_BOOTSTRAP="${EXTERNAL_BOOTSTRAP},"
  fi
  INTERNAL_BOOTSTRAP="${INTERNAL_BOOTSTRAP}citadel-kafka-${i}:9092"
  EXTERNAL_BOOTSTRAP="${EXTERNAL_BOOTSTRAP}citadel-kafka-${i}:9094"
done

for m in $(seq 1 "${MINION_COUNT}"); do
  MDIR="${ROOT_DIR}/configs/minion/minion-${m}"
  mkdir -p "${MDIR}"

  # NOTE: Internal paths still use /opt/opennms because that's where OpenNMS installs
  # But the hostname and visible names are citadelnms
  cat > "${MDIR}/org.opennms.minion.controller.cfg" << EOF
# Auto-generated by generate-compose.sh — CitadelNMS minion-${m}
location = ${MINION_LOCATION:-Default}
id = minion-${m}
http-url = http://citadelnms-host:8980/opennms
EOF

  cat > "${MDIR}/org.opennms.core.ipc.sink.kafka.cfg" << EOF
# Auto-generated by generate-compose.sh — CitadelNMS minion-${m} sink
bootstrap.servers=${EXTERNAL_BOOTSTRAP}
acks=1
linger.ms=5
batch.size=16384
compression.type=none
EOF

  cat > "${MDIR}/org.opennms.core.ipc.rpc.kafka.cfg" << EOF
# Auto-generated by generate-compose.sh — CitadelNMS minion-${m} rpc
bootstrap.servers=${EXTERNAL_BOOTSTRAP}
single-topic=false
EOF
  chmod -R a+rX "${MDIR}"
done

chmod -R a+rX "${ROOT_DIR}/configs/minion/featuresBoot.d"

# Update CitadelNMS kafka.properties
cat > "${ROOT_DIR}/configs/opennms/opennms.properties.d/kafka.properties" << EOF
# Auto-generated by generate-compose.sh — CitadelNMS Kafka Configuration
org.opennms.core.ipc.kafka.bootstrap.servers=${INTERNAL_BOOTSTRAP}
org.opennms.core.ipc.kafka.producer.acks=1
org.opennms.core.ipc.kafka.producer.linger.ms=5
org.opennms.core.ipc.kafka.producer.batch.size=16384
org.opennms.core.ipc.kafka.producer.compression.type=none
org.opennms.core.ipc.kafka.producer.max.block.ms=30000
org.opennms.core.ipc.kafka.consumer.fetch.min.bytes=1
org.opennms.core.ipc.kafka.consumer.fetch.max.wait.ms=10
org.opennms.core.ipc.kafka.consumer.max.poll.records=500
org.opennms.core.ipc.kafka.sink.consumer.threads=10
org.opennms.core.ipc.kafka.partitions=${KAFKA_PARTITIONS:-12}
EOF

chmod 644 "${ROOT_DIR}/configs/opennms/opennms.properties.d"/*.properties

# --- Generate docker-compose.yml ---------------------------------------------
echo "[3/3] Generating docker-compose.yml..."

COMPOSE_FILE="${ROOT_DIR}/docker-compose.yml"

cat > "${COMPOSE_FILE}" << 'HEADER'
######################################################################
# docker-compose.yml — CitadelNMS Multi-Container Architecture
# AUTO-GENERATED by scripts/generate-compose.sh
# Do not edit manually — re-run the script to regenerate.
######################################################################

networks:
  net-citadelnms:
    driver: bridge
    ipam:
      config:
        - subnet: 172.20.0.0/24
  net-kafka:
    driver: bridge
    ipam:
      config:
        - subnet: 172.21.0.0/24
  net-minion:
    driver: bridge
    ipam:
      config:
        - subnet: 172.22.0.0/24

volumes:
  citadel-postgres-data:
  citadelnms-data:
  citadelnms-logs:
HEADER

# Kafka volume declarations
for i in $(seq 1 "${BROKER_COUNT}"); do
  echo "  citadel-kafka-${i}-data:" >> "${COMPOSE_FILE}"
done

cat >> "${COMPOSE_FILE}" << 'SERVICES_HEADER'

services:

  # ── POSTGRESQL DATABASE ─────────────────────────────────────────
  citadel-postgres:
    image: postgres:${POSTGRES_VERSION:-15}
    container_name: citadel-postgres
    hostname: citadel-postgres
    networks: [net-citadelnms]
    environment:
      POSTGRES_DB: ${POSTGRES_DB:-citadelnms}
      POSTGRES_USER: ${POSTGRES_USER:-citadel}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-citadel123}
    volumes:
      - citadel-postgres-data:/var/lib/postgresql/data
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-citadel}"]
      interval: 10s
      timeout: 5s
      retries: 10
    deploy:
      resources:
        limits: {cpus: '1.0', memory: 2G}

  # ── CITADELNMS CORE ─────────────────────────────────────────────
  citadelnms-core:
    image: citadelnms:latest
    container_name: citadelnms-core
    hostname: citadelnms-core
    networks: [net-citadelnms, net-kafka]
    ports:
      - "8980:8980"
      - "8101:8101"
    depends_on:
      citadel-postgres:
        condition: service_healthy
    environment:
      TZ: UTC
      POSTGRES_HOST: citadel-postgres
      POSTGRES_PORT: 5432
      POSTGRES_USER: ${POSTGRES_USER:-citadel}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-citadel123}
      DISABLE_CONFD: "true"
      JAVA_OPTS: >-
        -Xmx${CITADELNMS_HEAP:-4G} -Xms${CITADELNMS_HEAP:-4G}
        -XX:+UseG1GC -Djava.net.preferIPv4Stack=true
    volumes:
      - citadelnms-data:/opennms-data
      - citadelnms-logs:/opt/opennms/logs
      - ./configs/opennms/opennms.properties.d:/opt/opennms/etc/opennms.properties.d
    command: ["-s"]
    user: "0:0"
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-sf", "http://localhost:8980/opennms/login.jsp"]
      interval: 30s
      timeout: 10s
      retries: 15
      start_period: 5m
    deploy:
      resources:
        limits: {cpus: '4.0', memory: 5G}

SERVICES_HEADER

# Kafka broker services
for i in $(seq 1 "${BROKER_COUNT}"); do
  HOST_INTERNAL_PORT=$((9000 + (i - 1) * 100 + 92))
  HOST_EXTERNAL_PORT=$((9000 + (i - 1) * 100 + 94))

  cat >> "${COMPOSE_FILE}" << EOF
  # ── CITADEL KAFKA BROKER ${i} ──────────────────────────────────
  citadel-kafka-${i}:
    image: apache/kafka:\${KAFKA_VERSION:-3.7.0}
    container_name: citadel-kafka-${i}
    hostname: citadel-kafka-${i}
    networks: [net-kafka, net-minion]
    ports:
      - "${HOST_INTERNAL_PORT}:9092"
      - "${HOST_EXTERNAL_PORT}:9094"
    volumes:
      - citadel-kafka-${i}-data:/var/lib/kafka/data
      - ./configs/kafka/broker-${i}.properties:/etc/kafka/broker.properties
    environment:
      KAFKA_CLUSTER_ID: \${KAFKA_CLUSTER_ID:-MkU3OEVBNTcwNTJENDM2Qg}
    command:
      - /bin/bash
      - -c
      - |
        if [ ! -f /var/lib/kafka/data/meta.properties ]; then
          echo "[CitadelNMS-Kafka] Formatting KRaft storage..."
          /opt/kafka/bin/kafka-storage.sh format \\
            --cluster-id "\$\${KAFKA_CLUSTER_ID}" \\
            --config /etc/kafka/broker.properties
        else
          echo "[CitadelNMS-Kafka] Storage already formatted."
        fi
        exec /opt/kafka/bin/kafka-server-start.sh /etc/kafka/broker.properties
    restart: unless-stopped
    deploy:
      resources:
        limits: {cpus: '2.0', memory: 2G}

EOF
done

# Minion services
DEPENDS_ON_KAFKA=""
for i in $(seq 1 "${BROKER_COUNT}"); do
  DEPENDS_ON_KAFKA="${DEPENDS_ON_KAFKA}      - citadel-kafka-${i}"$'\n'
done
DEPENDS_ON_KAFKA="${DEPENDS_ON_KAFKA%$'\n'}"

for m in $(seq 1 "${MINION_COUNT}"); do
  cat >> "${COMPOSE_FILE}" << EOF
  # ── CITADEL MINION ${m} ────────────────────────────────────────
  citadel-minion-${m}:
    image: citadelnms/minion:\${MINION_VERSION:-latest}
    container_name: citadel-minion-${m}
    hostname: citadel-minion-${m}
    networks: [net-minion]
    extra_hosts:
      - "citadelnms-host:host-gateway"
    volumes:
      - ./configs/minion/minion-${m}:/opt/minion-etc-overlay
      - ./configs/minion/featuresBoot.d:/opt/minion/etc/featuresBoot.d
    environment:
      MINION_LOCATION: \${MINION_LOCATION:-Default}
      MINION_ID: minion-${m}
      JAVA_OPTS: "-Xmx\${MINION_HEAP:-2G} -Xms\${MINION_HEAP:-2G}"
    depends_on:
EOF
  printf "%s\n" "${DEPENDS_ON_KAFKA}" >> "${COMPOSE_FILE}"
  cat >> "${COMPOSE_FILE}" << EOF
    restart: unless-stopped
    deploy:
      resources:
        limits: {cpus: '2.0', memory: 2500M}

EOF
done

echo ""
echo "============================================================"
echo " CitadelNMS docker-compose.yml generated successfully!"
echo ""
echo " Next steps:"
echo "   1. Build/tag CitadelNMS images (see README)"
echo "   2. docker compose up -d"
echo "   3. docker compose ps"
echo "============================================================"


### **4. Updated build-images.sh**

#!/usr/bin/env bash
# =============================================================================
# build-images.sh — Build CitadelNMS Docker images with rebranded labels
# =============================================================================
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
ROOT_DIR="$(dirname "$SCRIPT_DIR")"

echo "============================================================"
echo " Building CitadelNMS Docker Images from Dockerfiles"
echo "============================================================"

# Build CitadelNMS Horizon
echo "[1/3] Building CitadelNMS Horizon..."
docker build \
  -f "${ROOT_DIR}/dockerfiles/Dockerfile.horizon" \
  -t citadelnms:latest \
  "${ROOT_DIR}/dockerfiles"

# Build CitadelNMS Minion
echo "[2/3] Building CitadelNMS Minion..."
docker build \
  -f "${ROOT_DIR}/dockerfiles/Dockerfile.minion" \
  -t citadelnms/minion:latest \
  "${ROOT_DIR}/dockerfiles"

# Clean up old OpenNMS images (optional)
echo "[3/3] Cleaning up upstream images..."
docker rmi opennms/horizon:35.0.3 2>/dev/null || true
docker rmi opennms/minion:35.0.3 2>/dev/null || true

echo ""
echo "============================================================"
echo " CitadelNMS images built successfully:"
echo "============================================================"
docker images | grep citadel
echo ""
echo "Verify labels:"
echo "  docker inspect citadelnms:latest --format='{{json .Config.Labels}}' | jq"
echo "============================================================"
